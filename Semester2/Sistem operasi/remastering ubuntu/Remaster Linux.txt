INSTALL VSCODE
1. wget -qO- https://packages.microsoft.com/keys/microsoft.asc | gpg --dearmor > microsoft.gpg
2. install -o root -g root -m 644 microsoft.gpg /etc/apt/trusted.gpg.d/
3. sh -c 'echo "deb [arch=amd64] https://packages.microsoft.com/repos/code stable main" > /etc/apt/sources.list.d/vscode.list'
4. apt update
5. apt install code -y

INSTALL ANDROID STUDIO
